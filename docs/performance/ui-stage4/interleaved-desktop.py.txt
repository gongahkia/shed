from pathlib import Path
import hashlib, importlib.util, json, os, platform, queue, subprocess, tempfile, threading, time
repo=Path('/Users/gabrielongtemasek/Desktop/personal/shed')
out=Path('/tmp/shed-ui-stage4-typography')
spec=importlib.util.spec_from_file_location('original_benchmark',repo/'scripts/benchmark-editor.py')
original=importlib.util.module_from_spec(spec)
spec.loader.exec_module(original)
java=Path(subprocess.check_output([str(repo/'scripts/resolve-toolchain.sh'),'home'],text=True).strip())/'bin/java'
probe=repo/'src/benchmark/java/shed/EditorWorkflowProbe.java'
jars={'baseline':Path('/tmp/shed-ui-stage4-baseline/shed-2.0.0.jar'),'final':out/'shed.jar'}
order=['baseline','final','final','baseline','baseline','final','final','baseline','baseline','final']
samples={key:[] for key in jars}
commands=[]
with tempfile.TemporaryDirectory(prefix='shed-interleaved-') as temporary:
 work=Path(temporary)
 fixture=work/'fixture.txt'
 fixture.write_text(''.join(f'line {i:06d} alpha beta gamma delta epsilon zeta eta theta iota kappa\n' for i in range(20000)))
 for name,jar in jars.items():
  classes=work/name
  classes.mkdir()
  subprocess.run([str(java.parent/'javac'),'-cp',str(jar),'-d',str(classes),str(probe)],check=True)
 for index,name in enumerate(order):
  home=work/f'home-{index}'
  (home/'.shed').mkdir(parents=True)
  (home/'.shed/config.toml').write_text('schema_version = 1\n"keymap.profile" = "plain"\n"session.restore.on.start" = false\n')
  command=[str(java),'-Xms64m','-Xmx512m','-Djava.awt.headless=false',f'-Duser.home={home}','-cp',os.pathsep.join([str(work/name),str(jars[name])]),'shed.EditorWorkflowProbe',str(fixture)]
  commands.append({'variant':name,'command':command})
  print('START',index+1,name,flush=True)
  lines=queue.Queue()
  started=time.time()*1000
  process=subprocess.Popen(command,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
  def drain():
   for line in process.stdout:lines.put(line)
   lines.put(None)
  reader=threading.Thread(target=drain,daemon=True)
  reader.start()
  sample={}
  logs=[]
  try:
   while True:
    line=lines.get(timeout=120)
    if line is None:break
    logs.append(line.rstrip())
    if line.startswith('SHED_WINDOW '):sample['window_ms']=float(line.split()[1])-started
    elif line.startswith('SHED_SOURCE '):sample['source_ready_ms']=float(line.split()[1])-started
    elif line.startswith('SHED_READY '):
     sample['startup_ms']=float(line.split()[1])-started
     try:sample['rss_mib']=int(subprocess.check_output(['ps','-o','rss=','-p',str(process.pid)],text=True).strip())/1024
     except (ValueError,subprocess.SubprocessError,OSError):sample['rss_mib']=None
    elif line.startswith('SHED_RESULT '):sample.update(json.loads(line.removeprefix('SHED_RESULT ')))
   if process.wait(timeout=10)!=0 or 'edit_paint_ms' not in sample:raise RuntimeError('Probe failed:\n'+'\n'.join(logs))
  finally:
   if process.poll() is None:process.kill();process.wait()
   reader.join(timeout=5)
  if sample['language_servers']!=0:raise RuntimeError('Unexpected language server')
  samples[name].append(sample)
  (out/'interleaved-progress.json').write_text(json.dumps({'order':order,'completed':index+1,'runs':samples,'commands':commands},indent=2)+'\n')
  print('DONE',index+1,name,'startup',round(sample['startup_ms']),flush=True)
 for name,jar in jars.items():
  metrics={}
  for metric,budget in original.BUDGETS.items():
   values=[]
   for sample in samples[name]:
    value=sample.get(metric)
    values.extend(value if isinstance(value,list) else [] if value is None else [value])
   metrics[metric]={'p50':original.percentile(values,.50),'p95':original.percentile(values,.95),'budget':budget,'pass':original.percentile(values,.95)<=budget,'samples':len(values)} if values else {'unavailable':True}
  report={'label':'ui-stage4-interleaved-'+name,'measured_at_utc':time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()),'jar_sha256':hashlib.sha256(jar.read_bytes()).hexdigest(),'probe_sha256':hashlib.sha256(probe.read_bytes()).hexdigest(),'machine':platform.platform(),'cpu_count':os.cpu_count(),'java_heap':'64 MiB initial / 512 MiB maximum','fixture':{'bytes':fixture.stat().st_size,'lines':20000,'picker_paths':50000},'method':'Unmodified EditorWorkflowProbe and original budgets/percentile function; five fresh JVMs per artifact, same fixture/config/heap. Counterbalanced AB BA AB BA AB process order. No concurrent builds/tests; no OS cache purge.','order':order,'metrics':metrics,'runs':samples[name]}
  (out/('interleaved-'+name+'-desktop.json')).write_text(json.dumps(report,indent=2)+'\n')
 print('INTERLEAVED_COMPLETE',flush=True)
